---- Context ----
public bool IsFireInput = false;        // 발사 입력
public bool IsAiming = false;           // 에임중 여부
public bool CanFire = false;            // 발사 가능 유무
public bool IsRecoil = false;           // 반동중인지 확인

public int CurrentCapacity;             // 현재 장탄수
public int MaxCapacity;                 // 최대 장탄수
public int AmmoRemaining;               // 남은 탄약수
public int MaxRemaining;                // 최대 탄약수

public bool IsReloadInput = false;      // 재장전 입력
public bool IsReloading = false;        // 재장전중인지 확인

--- 트랜지션 ----
Idle -> Aim : 발사입력이 있을때
Recoil -> Aim : 반동중이 아니고, 발사중이 아닐때

Idle -> Reload : 현재 장탄수가 최대 장탄수 미만이고, 재장전 입력이 있으며, 남은 탄약수가 0 이상일때
Aim -> Reload : (현재 장탄수가 없으며 남은 탄약수가 0 이상일때) or (현재 장탄수가 최대 장탄수 미만이고, 재장전 입력이 있으며, 남은 				탄약수가 0 이상일때)

Aim -> Idle : 발사 입력 n초간 입력 x
Relaod -> Idle : 재장전중이 아니고 입력이 없을때, 에임중이 아닐때

Aim -> Fire : 발사 입력이 있을때

Fire -> Recoil : 발사중일때

--- 로직 ---
IdleState : 
{
 StateEnter 		{ IsAiming = false; CanFire = true; IsRecoil = false; IsReload = false;}
 StateUpdate 	{ }
 StateExit 		{ }
}

AimState : 
{
 StateEnter 		{ CanFire = true; IsRecoil = false; }
 StateUpdate 	{ n초 타이머후 IsAiming = false; }
 StateExit 		{ }
}

FireState : 
{
 StateEnter 		{ CanFire = false; }
 StateUpdate	{ }
 StateExit 		{ }
}

RecoilState :
{
 StateEnter 		{ CanFire = false;  IsRecoil = true; }
 StateUpdate 	{ }
 StateExit 		{ }
}

ReloadState :
{
 StateEnter 		{ IsReloading = true }
 StateUpdate 	{ }
 StateExit 		{ }
}
